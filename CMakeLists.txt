cmake_minimum_required(VERSION 3.16)
project(gibus VERSION 0.1.0 LANGUAGES C)

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    set(_gibus_top ON)
else()
    set(_gibus_top OFF)
endif()
option(GIBUS_TESTS "Build gibustest" ${_gibus_top})

add_library(gibus STATIC src/gibus_msg.c)
target_include_directories(gibus PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(gibus PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)

set(GIBUS_CONNECTION OFF)
if(CMAKE_SYSTEM_NAME MATCHES "^(Linux|FreeBSD|OpenBSD|NetBSD|DragonFly)$")
    set(GIBUS_CONNECTION ON)
    target_sources(gibus PRIVATE src/gibus_conn.c)
    set(THREADS_PREFER_PTHREAD_FLAG ON)
    find_package(Threads REQUIRED)
    target_link_libraries(gibus PUBLIC Threads::Threads)
endif()

if(GIBUS_TESTS)
    add_executable(gibustest tests/gibustest.c)
    target_include_directories(gibustest PRIVATE src)
    target_link_libraries(gibustest PRIVATE gibus)
    if(GIBUS_CONNECTION)
        target_compile_definitions(gibustest PRIVATE GIBUSTEST_LIVE)
    endif()
    set_target_properties(gibustest PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON
                                               C_EXTENSIONS OFF)
endif()

if(MSVC)
    target_compile_options(gibus PRIVATE /W4 /we4013)
    if(GIBUS_TESTS)
        target_compile_options(gibustest PRIVATE /W4 /we4013)
    endif()
else()
    target_compile_options(gibus PRIVATE -Wall -Wextra)
    if(GIBUS_TESTS)
        target_compile_options(gibustest PRIVATE -Wall -Wextra)
    endif()
endif()
